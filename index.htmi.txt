    <script>
        const SYSTEM_SECRET = "Sadreen_Secure_2026";
        let centerLocation = { lat: 33.3889, lng: 44.4522, radius: 50 };
        let currentUser = null;
        let userCurrentPosition = null;

        const defaultUsers = [
            { id: 'admin_1', name: 'مدير النظام (أبو سيف)', username: 'Sadreen', password: '197111', role: 'admin', salary: 0 },
            { id: 'emp_1', name: 'أحمد علي', username: 'ahmed', password: '1234', role: 'employee', salary: 600000 },
            { id: 'emp_2', name: 'علي حسين', username: 'ali', password: '1234', role: 'employee', salary: 550000 }
        ];

        function initData() {
            if (!localStorage.getItem('sadreen_users')) localStorage.setItem('sadreen_users', JSON.stringify(defaultUsers));
            if (!localStorage.getItem('sadreen_attendance')) localStorage.setItem('sadreen_attendance', JSON.stringify([]));
            if (!localStorage.getItem('sadreen_leaves')) localStorage.setItem('sadreen_leaves', JSON.stringify([]));
            if (localStorage.getItem('sadreen_gps_center')) centerLocation = JSON.parse(localStorage.getItem('sadreen_gps_center'));
        }
        initData();

        function generateHash(str) { return CryptoJS.SHA256(str + SYSTEM_SECRET).toString(); }

        function calculateDistance(lat1, lon1, lat2, lon2) {
            const R = 6371e3;
            const ?1 = lat1 * Math.PI / 180, ?2 = lat2 * Math.PI / 180;
            const ?? = (lat2 - lat1) * Math.PI / 180, ?? = (lon2 - lon1) * Math.PI / 180;
            const a = Math.sin(??/2)**2 + Math.cos(?1) * Math.cos(?2) * Math.sin(??/2)**2;
            return Math.round(R * (2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a))));
        }

        function handleLogin(e) {
            e.preventDefault();
            const u = document.getElementById('loginUsername').value.trim();
            const p = document.getElementById('loginPassword').value.trim();
            const users = JSON.parse(localStorage.getItem('sadreen_users'));
            const found = users.find(x => x.username.toLowerCase() === u.toLowerCase() && x.password === p);

            if (found) {
                currentUser = found;
                document.getElementById('loginError').classList.add('hidden');
                document.getElementById('loginScreen').classList.add('hidden');
                document.getElementById('dashboardScreen').classList.remove('hidden');
                document.getElementById('userHeaderInfo').classList.remove('hidden');
                document.getElementById('headerUserName').innerText = currentUser.name;
                document.getElementById('headerUserRole').innerText = currentUser.role === 'admin' ? 'مدير النظام' : 'موظف';
                setupDashboard();
                trackGps();
            } else {
                document.getElementById('loginError').classList.remove('hidden');
            }
        }

        function logout() {
            currentUser = null;
            document.getElementById('dashboardScreen').classList.add('hidden');
            document.getElementById('userHeaderInfo').classList.add('hidden');
            document.getElementById('loginScreen').classList.remove('hidden');
        }

        function setupDashboard() {
            if (currentUser.role === 'admin') {
                document.getElementById('adminGpsBanner').classList.remove('hidden');
                document.getElementById('adminPanel').classList.remove('hidden');
                document.getElementById('coordsDisplay').innerText = `خط العرض: ${centerLocation.lat} | خط الطول: ${centerLocation.lng}`;
                renderEmployees();
                renderLogs();
                renderLeaves();
            } else {
                document.getElementById('adminGpsBanner').classList.add('hidden');
                document.getElementById('adminPanel').classList.add('hidden');
            }
        }

        function trackGps() {
            if ("geolocation" in navigator) {
                navigator.geolocation.watchPosition(pos => {
                    userCurrentPosition = { lat: pos.coords.latitude, lng: pos.coords.longitude };
                    const dist = calculateDistance(userCurrentPosition.lat, userCurrentPosition.lng, centerLocation.lat, centerLocation.lng);
                    const txt = document.getElementById('gpsDistanceText');
                    if (dist <= centerLocation.radius) {
                        txt.innerHTML = `<span class="text-emerald-400 font-bold">أنت داخل نطاق المركز (المسافة: ${dist} متر)</span>`;
                    } else {
                        txt.innerHTML = `<span class="text-red-400 font-bold">أنت بعيد عن المركز (المسافة: ${dist} متر - الحد 50م)</span>`;
                    }
                }, () => {
                    document.getElementById('gpsDistanceText').innerText = "يرجى تفعيل الـ GPS والسماح للمتصفح بالوصول للموقع.";
                }, { enableHighAccuracy: true });
            }
        }

        function setCurrentLocationAsCenter() {
            if (userCurrentPosition) {
                centerLocation.lat = userCurrentPosition.lat;
                centerLocation.lng = userCurrentPosition.lng;
                localStorage.setItem('sadreen_gps_center', JSON.stringify(centerLocation));
                document.getElementById('coordsDisplay').innerText = `خط العرض: ${centerLocation.lat} | خط الطول: ${centerLocation.lng}`;
                alert('تم اعتماد موقعك الحالي كمقر للمركز بنجاح!');
            } else {
                alert('جاري تحديد موقعك... انتظر قليلاً.');
            }
        }

        function registerAttendance(type) {
            if (!userCurrentPosition) {
                alert("يرجى تفعيل الـ GPS أولاً!");
                return;
            }
            const dist = calculateDistance(userCurrentPosition.lat, userCurrentPosition.lng, centerLocation.lat, centerLocation.lng);
            if (dist > centerLocation.radius) {
                alert(`عذراً، المسافة عن المركز هي ${dist} متر. يجب أن تكون أقل من 50 متراً.`);
                return;
            }
            const logs = JSON.parse(localStorage.getItem('sadreen_attendance'));
            const now = new Date().toLocaleString('ar-IQ');
            logs.unshift({
                id: Date.now(),
                userName: currentUser.name,
                type: type === 'checkIn' ? 'حضور' : 'انصراف',
                timestamp: now,
                distance: dist,
                hash: generateHash(`${currentUser.id}_${type}_${now}`)
            });
            localStorage.setItem('sadreen_attendance', JSON.stringify(logs));
            alert(`تم تسجيل ${type === 'checkIn' ? 'الحضور' : 'الانصراف'} بنجاح!`);
            if (currentUser.role === 'admin') renderLogs();
        }

        function renderEmployees() {
            const users = JSON.parse(localStorage.getItem('sadreen_users'));
            const tbody = document.getElementById('employeesTableBody');
            tbody.innerHTML = users.map(u => `
                <tr class="hover:bg-slate-800/40">
                    <td class="p-4 font-bold text-white">${u.name}</td>
                    <td class="p-4 font-mono text-slate-400">${u.username}</td>
                    <td class="p-4 font-mono text-amber-400">${u.password}</td>
                    <td class="p-4 font-bold text-emerald-400">${u.salary.toLocaleString()} د.ع</td>
                    <td class="p-4 text-center">
                        ${u.role !== 'admin' ? `<button onclick="deleteEmp('${u.id}')" class="text-red-400"><i class="fa-solid fa-trash"></i></button>` : '-'}
                    </td>
                </tr>
            `).join('');
        }

        function renderLogs() {
            const logs = JSON.parse(localStorage.getItem('sadreen_attendance'));
            document.getElementById('attendanceLogsTableBody').innerHTML = logs.map(l => `
                <tr>
                    <td class="p-4 font-mono text-slate-400">${l.timestamp}</td>
                    <td class="p-4 font-bold text-white">${l.userName}</td>
                    <td class="p-4"><span class="px-2 py-1 rounded text-xs font-bold ${l.type === 'حضور' ? 'bg-emerald-500/10 text-emerald-400' : 'bg-amber-500/10 text-amber-400'}">${l.type}</span></td>

                </tr>
            `).join('');
        }

        function renderLeaves() {
            const leaves = JSON.parse(localStorage.getItem('sadreen_leaves'));
            document.getElementById('leavesTableBody').innerHTML = leaves.map(lv => `
                <tr>
                    <td class="p-4 font-bold text-white">${lv.userName}</td>
                    <td class="p-4">${lv.reason}</td>
                    <td class="p-4 font-mono text-slate-400">${lv.startDate}</td>
                    <td class="p-4 font-mono text-slate-400">${lv.endDate}</td>
                    <td class="p-4"><span class="px-2 py-1 rounded text-xs ${lv.status === 'مقبول' ? 'bg-emerald-500/10 text-emerald-400' : lv.status === 'مرفوض' ? 'bg-red-500/10 text-red-400' : 'bg-amber-500/10 text-amber-400'}">${lv.status}</span></td>
                    <td class="p-4 text-center">
                        ${lv.status === 'قيد الانتظار' ? `<button onclick="updateLeave('${lv.id}', 'مقبول')" class="px-2 py-1 bg-emerald-600 text-white rounded text-xs">قبول</button> <button onclick="updateLeave('${lv.id}', 'مرفوض')" class="px-2 py-1 bg-red-600 text-white rounded text-xs">رفض</button>` : '-'}
                    </td>
                </tr>
            `).join('');
        }

        function openAddEmployeeModal() { document.getElementById('employeeModal').classList.remove('hidden'); }
        function closeEmployeeModal() { document.getElementById('employeeModal').classList.add('hidden'); }

        function saveEmployee(e) {
            e.preventDefault();
            const users = JSON.parse(localStorage.getItem('sadreen_users'));
            users.push({
                id: 'emp_' + Date.now(),
                name: document.getElementById('modalEmpName').value,
                username: document.getElementById('modalEmpUser').value,
                password: document.getElementById('modalEmpPass').value,
                role: 'employee',
                salary: parseFloat(document.getElementById('modalEmpSalary').value)
            });
            localStorage.setItem('sadreen_users', JSON.stringify(users));
            closeEmployeeModal();
            renderEmployees();
            alert('تمت إضافة الموظف بنجاح!');
        }

        function deleteEmp(id) {
            if (confirm('هل أنت متأكد من الحذف؟')) {
                let users = JSON.parse(localStorage.getItem('sadreen_users')).filter(x => x.id !== id);
                localStorage.setItem('sadreen_users', JSON.stringify(users));
                renderEmployees();
            }
        }

        function handleLeaveSubmit(e) {
            e.preventDefault();
            const leaves = JSON.parse(localStorage.getItem('sadreen_leaves'));
            leaves.unshift({
                id: Date.now(),
                userName: currentUser.name,
                reason: document.getElementById('leaveReason').value,
                startDate: document.getElementById('leaveStartDate').value,
                endDate: document.getElementById('leaveEndDate').value,
                status: 'قيد الانتظار'
            });
            localStorage.setItem('sadreen_leaves', JSON.stringify(leaves));
            alert('تم إرسال طلب الإجازة بنجاح!');
            document.getElementById('leaveReason').value = '';
        }

        function updateLeave(id, status) {
            let leaves = JSON.parse(localStorage.getItem('sadreen_leaves'));
            let target = leaves.find(x => x.id == id);
            if (target) { target.status = status; localStorage.setItem('sadreen_leaves', JSON.stringify(leaves)); renderLeaves(); }
        }

        function exportAuditData() {
            const data = { users: JSON.parse(localStorage.getItem('sadreen_users')), attendance: JSON.parse(localStorage.getItem('sadreen_attendance')), leaves: JSON.parse(localStorage.getItem('sadreen_leaves')) };
            const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
            const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = 'Sadreen_Backup.json'; a.click();
        }

            ['employees', 'logs', 'leaves'].forEach(t => {

            });
            document.getElementById(`tabContent-${tab}`).classList.remove('hidden');
            document.getElementById(`tabBtn-${tab}`).classList.add('border-blue-500', 'text-blue-400');
        }
